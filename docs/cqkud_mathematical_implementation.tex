\begin{purposebox}
\textbf{Purpose.}  The solver reports a projected Ritz energy from its
Lanczos states.  The separate collected expression gives each state's
expectation as a sum of \(C_{n,k}\) terms, with the global factor absorbed.
\end{purposebox}

\guidesection{1}{Scale the Hamiltonian and define the operator}
Write the physical Hamiltonian as Pauli terms and scale it by their
coefficient one-norm:
\begin{equation}
 H=\sum_\ell h_\ell P_\ell,\qquad
 \Lambda=\sum_\ell|h_\ell|,\qquad
 \bar H=H/\Lambda .
 \label{eq:scale}
\end{equation}
The normalized reference \(\ket{\Phi_0}\) is the starting state.
For real \(\epsilon\), define
\begin{equation}
 U_\epsilon=\exp(-i\epsilon\bar H),\qquad
 X_\epsilon=\frac{U_\epsilon^\dagger-U_\epsilon}{2i}
            =\sin(\epsilon\bar H).
 \label{eq:filter}
\end{equation}
The circuits use \(iU_\epsilon\), whose Hermitian part is
\(X_\epsilon\); the collected terms use \(U_\epsilon\).
At \(\epsilon=0\), only the reference state survives.

\guidesection{2}{Measure moments and generate Lanczos states}
Start with \(\ket{q_0}=\ket{\Phi_0}\),
\(\ket{q_{-1}}=0\), and \(\beta_0=0\).  The two controlled-unitary
circuits measure
\begin{equation}
 \begin{aligned}
 \alpha_n&=\operatorname{Re}\bra{q_n}iU_\epsilon\ket{q_n}
           =\bra{q_n}X_\epsilon\ket{q_n},\\
 \mu_{2,n}&=\frac{1+\operatorname{Re}
             \bra{q_n}(iU_\epsilon)^2\ket{q_n}}{2}
             =\bra{q_n}X_\epsilon^2\ket{q_n},\\
 \beta_{n+1}&=\sqrt{\mu_{2,n}-\alpha_n^2-\beta_n^2},\qquad
 \ket{q_{n+1}}=\frac{X_\epsilon\ket{q_n}
     -\alpha_n\ket{q_n}-\beta_n\ket{q_{n-1}}}{\beta_{n+1}}.
 \end{aligned}
 \label{eq:lanczos}
\end{equation}
The code clips small negative \(\beta_{n+1}^2\) from roundoff and stops
at breakdown or the requested dimension.  For
\(0<|\epsilon|<10^{-3}\), it measures \(X_\epsilon\) and
\(X_\epsilon^2\) directly to avoid cancellation in the probabilities.

\begin{featurebox}{Polynomial form of a state}
The same states can be written with monic polynomials:
\begin{equation}
 \begin{gathered}
 p_{-1}(x)=0,\quad p_0(x)=1,\qquad
 p_{n+1}(x)=(x-\alpha_n)p_n(x)-\beta_n^2p_{n-1}(x),\\
 \Ncal_n=\prod_{j=1}^{n}\beta_j^2,\quad \Ncal_0=1,\qquad
 \ket{q_n}=\frac{p_n(X_\epsilon)\ket{\Phi_0}}{\sqrt{\Ncal_n}}.
 \end{gathered}
 \label{eq:polynomial}
\end{equation}
Because \(H\) commutes with \(X_\epsilon\), the individual state energy is
\(E_n^{\mathrm{state}}=
\bra{\Phi_0}H p_n(X_\epsilon)^2\ket{\Phi_0}/\Ncal_n\).
\end{featurebox}

In \texttt{run()}, \(\ket{q_n}=\sum_{m=0}^{n}c_{n,m}
T_m(Y_\epsilon)\ket{\Phi_0}\), with \(Y_\epsilon=X_\epsilon/s_\epsilon\).
Here \(s_\epsilon=|\epsilon|\) only for \(0<|\epsilon|<10^{-3}\);
otherwise it is one.  The Lanczos relation updates \(c_{n,m}\) using
\(yT_0(y)=T_1(y)\) and
\(yT_m(y)=[T_{m-1}(y)+T_{m+1}(y)]/2\).
Clenshaw evaluation constructs and normalizes each state.

\guidesection{3}{Project the Hamiltonian and report the Ritz energy}
Circuits measure \(S_{ij}=\langle q_i|q_j\rangle\) and
\(M_{ij}=\bra{q_i}\bar H\ket{q_j}\).  For each prefix of \(K\) states,
the analytic (\texttt{shots=None}) path reports
\begin{equation}
 E_K^{\mathrm{Ritz}}=\Lambda\lambda_{\min}(M_K),\qquad
 \kappa(S_K)=\frac{\lambda_{\max}(S_K)}{\lambda_{\min}(S_K)}.
 \label{eq:ritz}
\end{equation}
The condition number uses retained positive overlap eigenvalues, with no
``excess'' subtraction.  The code normalizes but does not reorthogonalize
the states, so \(S_K\) can differ from \(I\).  With finite shots it solves
the retained generalized problem \(M_K z=(E/\Lambda)S_K z\).
In exact arithmetic, the collected \(E_n^{\mathrm{state}}=\Lambda M_{nn}\)
is generally different from \(E_{n+1}^{\mathrm{Ritz}}\).

\guidesection{4}{Use the moments shown with the collected terms}
Let \(m_k=\bra{\Phi_0}HU_\epsilon^k\ket{\Phi_0}\).
For each term, use the same paired bracket as the collected-values PDF:
\begin{equation}
 \begin{aligned}
 \Bcal_0&=m_0=\bra{\Phi_0}H\ket{\Phi_0},\\
 \Bcal_k&=m_k^*+m_k
   =\bra{\Phi_0}H\bigl((U_\epsilon^\dagger)^k+
                         U_\epsilon^k\bigr)\ket{\Phi_0},
                         &&k>0\ \text{even},\\
 \Bcal_k&=m_k^*-m_k
   =\bra{\Phi_0}H\bigl((U_\epsilon^\dagger)^k-
                         U_\epsilon^k\bigr)\ket{\Phi_0},
                         &&k\ \text{odd}.
 \end{aligned}
 \label{eq:brackets}
\end{equation}
For eigenpairs \(\bar H\ket{\ell}=\lambda_\ell\ket{\ell}\)
and weights \(w_\ell=|\langle\ell|\Phi_0\rangle|^2\),
\(m_k=\Lambda\sum_\ell w_\ell\lambda_\ell
e^{-ik\epsilon\lambda_\ell}\).
Thus even \(\Bcal_k\) are real and odd \(\Bcal_k\) are imaginary.
The collected-values script converts the solver's phase-shifted moments
to these displayed \(U_\epsilon\) moments.

\guidesection{5}{Collect the matching $C$ coefficients}
Write \(p_n(x)^2=\sum_{r=0}^{2n}b_{n,r}x^r\).  First collect
the real binomial weights
\begin{equation}
 A_{n,k}=
 \sum_{\substack{r=k\\r-k\,\mathrm{even}}}^{2n}
 \frac{b_{n,r}}{2^r}\binom{r}{(r-k)/2},
 \qquad 0\leq k\leq2n.
 \label{eq:weights}
\end{equation}
Because \(X_\epsilon=(U_\epsilon^\dagger-U_\epsilon)/(2i)\),
the power expansion is
\begin{equation}
 p_n(X_\epsilon)^2=A_{n,0}I+
 \sum_{k=1}^{2n}A_{n,k}
 \bigl(i^k U_\epsilon^k+
       (-i)^k(U_\epsilon^\dagger)^k\bigr).
 \label{eq:powers}
\end{equation}
The coefficient of each displayed bracket, after multiplying
through by the global normalization, is
\begin{equation}
 \boxed{\displaystyle
 C_{n,k}=\frac{(-i)^k A_{n,k}}{\Ncal_n},
 \qquad
 E_n^{\mathrm{state}}=\bra{q_n}H\ket{q_n}
   =C_{n,0}\Bcal_0+C_{n,1}\Bcal_1+\cdots+
     C_{n,2n}\Bcal_{2n}.}
 \label{eq:final}
\end{equation}
For even \(k\), both \(C_{n,k}\) and \(\Bcal_k\) are real.
For odd \(k\), both are imaginary and their product is real.
If the original numerator and denominator are scaled by \(4^n\),
then \(D_n=4^n\Ncal_n\) and the displayed
\(C_{n,k}=[4^n(-i)^k A_{n,k}]/D_n\).
No denominator remains outside the final sum.

\begin{navybox}{First state: the same terms and moments}
For \(p_1(x)=x-\alpha_0\),
\begin{equation}
 \begin{gathered}
 C_{1,0}=\frac{\alpha_0^2+1/2}{\beta_1^2},\qquad
 C_{1,1}=\frac{i\alpha_0}{\beta_1^2},\qquad
 C_{1,2}=-\frac{1}{4\beta_1^2},\\
 E_1^{\mathrm{state}}=C_{1,0}\Bcal_0+C_{1,1}\Bcal_1+C_{1,2}\Bcal_2.
 \end{gathered}
 \label{eq:first}
\end{equation}
At \(\epsilon=0.5\), these are the same displayed values:
\begin{center}
\small
\begin{tabular}{c@{\qquad}r@{\qquad}r}
\(k\) & \(C_{1,k}\) & \(\Bcal_k\) (Hartree)\\
\hline
0 & \(569.0327900720766\) & \(-1.7139986422219549103\)\\
1 & \(-162.6272732871267\,i\) & \(0.53284571232561824927\,i\)\\
2 & \(-272.3790682842995\) & \(-3.2585459375096481863\)
\end{tabular}
\end{center}
Their unrounded sum is \(E_1^{\mathrm{state}}\simeq-1.10647786459087\)
Hartree.  This is not the \(K=2\) Ritz energy reported by \texttt{run()}.
\end{navybox}

\begin{purposebox}
\textbf{In one line:} use the displayed moments \(\Bcal_k\),
multiply each by its distributed coefficient \(C_{n,k}\), and sum
\(E_n^{\mathrm{state}}=\sum_{k=0}^{2n}C_{n,k}\Bcal_k\).
\end{purposebox}
